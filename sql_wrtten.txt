--86. Banking Scenario: Given CUSTOMER, ACCOUNT, and BANK_TRANSACTION, write a query to display customers who have at
--least one account whose balance is greater than the average account balance. 

select c.FIRST_NAME, a.ACCOUNT_TYPE, t.TXN_TYPE, a.available_balance
from FHB_CUSTOMER c join fhb_account a
on c.CUSTOMER_ID = a.customer_id
join fhb_transaction t
on a.ACCOUNT_ID = t.ACCOUNT_ID
where a.available_balance > (select avg(a.available_balance) from fhb_account a);
