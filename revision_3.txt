--Table 1 - SBI_CUSTOMER

Create table SBI_CUSTOMER(
customer_id number(5) constraint cust_id_pk PRIMARY KEY,
customer_name VARCHAR2(20) CONSTRAINT cust_name_nn NOT NULL,
phone number(10) CONSTRAINT cust_num_uni UNIQUE,
email VARCHAR2(20) constraint cust_mail_uni UNIQUE,
city VARCHAR2(15) CONSTRAINT cust_city_nn not null);

select * from SBI_CUSTOMER;

--
--SBI_Account Table: constraint prod_code_fk foreign key(product_code) references product(product_code)

create table SBI_ACCOUNT(account_id number constraint acc_id_pk PRIMARY KEY,
customer_id number(10), constraint cust_id_fk foreign key (customer_id) REFERENCES SBI_CUSTOMER(customer_id),
account_type VARCHAR2(10) CONSTRAINT acc_type_ck CHECK(account_type in ('SAVINGS', 'CURRENT')),
balance number(10) constraint bal_ck CHECK (balance>= 0),
open_date date constraint op_dt_nn NOT NULL);

--SBI_Transaction Table:

create table SBI_TRANSACTION (txn_id number constraint txn_id_pk PRIMARY KEY,
account_id number(10), constraint acc_id_fk FOREIGN KEY(account_id) references sbi_Account(account_id),
txn_type varchar2(10) CONSTRAINT txn_type_ck CHECK(txn_type in('DEBIT', 'CREDIT')),
amount number constraint am_ck CHECK (amount > 0),
txn_date date constraint txn_dyt_nn NOT NULL);
