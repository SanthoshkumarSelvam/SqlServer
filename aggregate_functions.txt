--Aggregate Functions

--count, sum, average, min, max

--count 

select * from employees;

select COUNT(*) from EMPLOYEES; -- Star counts null as a value

select count(salary), count(COMMISSION_PCT) from employees;

--sum

select sum(salary) from EMPLOYEES;

--average

select AVG(salary) from employees;

select round(AVG(salary)) from employees;

--min

select min(salary) from EMPLOYEES;

--max

select max(salary) from EMPLOYEES;

--group_by and having

select department_id, COUNT(salary), sum(SALARY) from employees GROUP by department_id;

select DEPARTMENT_ID, round(avg(SALARY)) from employees GROUP BY DEPARTMENT_ID;

select DEPARTMENT_ID, sum(SALARY) from employees GROUP by DEPARTMENT_ID;

select department_id, max(salary) from employees GROUP BY DEPARTMENT_ID;

select department_id, min(salary) from employees GROUP BY DEPARTMENT_ID;
